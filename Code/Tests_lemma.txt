def Random_lattice(n, p):
    return posets.RandomPoset(n, p).completion_by_cuts().canonical_label()

def Random_meetsemidistributive_lattice(n, p):
    while True:
        L = Random_lattice(n, p)
        if L.is_meet_semidistributive() and not L.is_join_semidistributive():
            return L

def Random_joinsemidistributive_lattice(n, p):
    while True:
        L = Random_lattice(n, p)
        if L.is_join_semidistributive() and not L.is_meet_semidistributive():
            return L

def Random_semidistributive_lattice(n, p):
    while True:
        L = Random_lattice(n, p)
        if L.is_semidistributive():
            return L

def transvals_from_CJL(L):
    G = {x: L.canonical_joinands(x) for x in L}
    E = {}
    for x, y in L.cover_relations():
        for j in G[y]:
            if not L.is_lequal(j, x):
                E[(x, y)] = j
                break
    U = {x: [E[(x, y)] for y in L.upper_covers(x)] for x in L}
    return [(x, y) for x in L for y in L if all(L.is_lequal(j, y) or j in U[y] for j in G[x])]
            


def test_conuclear_up(L, display = False):
    D = defaultdict(list)
    for x, y in L.relations():
        if is_conuclear(L, x, y):
            D[x].append(y)
    Errors = set()
    for x in D:
        l = D[x]
        for i, y in enumerate(l):
            for z in l[i+1:]:
                m = L.meet([y, z])
                if m not in l:
                    if display:
                        print(f"Meet of upper bounds fails: [{x}, {y}] and [{x}, {z}] are conuclear intervals, but [{x}, {m}] is not")
                    Errors.add("upmeet_not_conuclear")
                if not is_nuclear(L, x, m):
                    if display:
                        print(f"Meet of upper bounds fails: [{x}, {y}] and [{x}, {z}] are conuclear intervals, but [{x}, {m}] is not nuclear")
                    Errors.add("upmeet_not_nuclear")
                j = L.join([y, z])
                if j not in l:
                    if display:
                        print(f"Join of upper bounds fails: [{x}, {y}] and [{x}, {z}] are conuclear intervals, but [{x}, {j}] is not")
                    Errors.add("upjoin_not_conuclear")
                if not is_nuclear(L, x, j):
                    if display:
                        print(f"Join of upper bounds fails: [{x}, {y}] and [{x}, {z}] are conuclear intervals, but [{x}, {j}] is not nuclear")
                    Errors.add("upjoin_not_nuclear")
    return Errors

def Test_face_join_transvals(L):
    L2 = semidistributive_transvals(L)
    for a, b in L2:
        for c, d in L2:
            if b == d:
                y = L.join([a, c])
                x = L.meet([b, y])
                t = L.join([L.meet([a,b]), L.meet([c,b])])
                x2 = popdown(L, y, t)
                if x != x2:
                    print(a,b,c,d,x,y,x2)

def Test_face_join_transvals2(L):
    for a, b, c in Tuples(L, 3):
        a2, b2 = L.meet([a,c]), L.meet([b,c])
        if not L.is_lequal(c, a) and not L.is_lequal(c, b) and is_nuclear(L, a2, a) and is_nuclear(L, b2, b):
            z = L.join([a,b])
            x, y = L.join([a2, b2]), L.meet([z,c])
            if popdown(L, z, x) != y:
                print(a,b,c,a2,b2,x,y,z)
                return False
    return True
